cmake_minimum_required(VERSION 3.24)
project(SCSOStormCall VERSION 3.0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
# Dependencies restored from the corresponding published release source.
set(COMMONLIBSSE_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../dependencies/CommonLibSSE-NG-10.0.1" CACHE PATH "Matching CommonLibSSE-NG source")
set(COMMONLIBSSE_INSTALL_PREFIX "${CMAKE_CURRENT_SOURCE_DIR}/../build/commonlib-installed" CACHE PATH "Matching source-built dependencies")
list(PREPEND CMAKE_PREFIX_PATH "${COMMONLIBSSE_INSTALL_PREFIX}")

find_package(fmt CONFIG REQUIRED)
find_package(spdlog CONFIG REQUIRED)
find_path(RAPIDCSV_INCLUDE_DIR rapidcsv.h REQUIRED)
find_package(directxtk CONFIG REQUIRED)
find_package(CommonLibSSE 10.0.1 CONFIG REQUIRED)
add_library(SCSOStormCall SHARED src/skse/main.cpp)
target_compile_definitions(SCSOStormCall PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(SCSOStormCall PRIVATE ${RAPIDCSV_INCLUDE_DIR})
target_link_libraries(SCSOStormCall PRIVATE CommonLibSSE::CommonLibSSE)
target_precompile_headers(SCSOStormCall PRIVATE <RE/Skyrim.h> <SKSE/SKSE.h>)
set_target_properties(SCSOStormCall PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/Data/SKSE/Plugins")
enable_testing()
add_executable(scso_repeat_queue_test tests/repeat_queue.cpp)
target_compile_options(scso_repeat_queue_test PRIVATE /UNDEBUG)
add_test(NAME scso_repeat_queue_regression COMMAND scso_repeat_queue_test)
